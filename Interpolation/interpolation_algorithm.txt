DEFINE structure Point {
    x, y
}

BEGIN main ()
    for generating a polynomial of order N
    input: take input using argc and **argv the following things,
                    M - no of points (we will generate 2*M points on the curve and use M to generate the Lagrange and Newton's polynomials and the other N to calculate the error between original and the other M points that are generated using Lagrange and Newton's polynomial)
                    N - order of the polynomial 
                    stdev - standard deviation for the gaussian noise that we will add to the 2*M points to simulate small error in them
                    take all the coefficients a_0, a_1, a_2, a_3, .... , a_n and store them in an array called coefficients[0 .. N]

    check if the user has given all the n+1 coefficients if not then throw an error and exit

    starting value = pick random number between -5 and 5 //this is a double 
    spacing = 2 
    //we will use the staring value for x and spacing to generate 2*M evenly spaced points 
    
    arr points[0 .. 2*M - 1] //array to store 2*M points as Point structure 
    generate_points(points, starting value, spacing, stdev, coefficients) //this will generate 2*M points and also adds the gaussian error term

    arr lagrange_points[0 .. M-1] //array to store the calculated lagrange points for error calculation
    arr newton_points[0 .. M-1] //array to store the calculated newton points for error calculation

    Lagrange(points, lagrange_points)
    Newton(points, newton_points)

    calc_root_mean_square_error(points, lagrange_points, newton_points) //calculates error and prints it to the screen
END main

BEGIN generate_points (input: arr, start, step, stdev, coefficients //take arrays as reference as we are going to change the values of the original array)
    
    BEGIN for (i = 0, i < 2*M, i++) 
        arr[i].x = start + step * i 
        arr[i].y = polynomial(point.x, coefficients) + gaussian_error(stdev)
    END for 
END generate_points

BEGIN polynomial (input: x, coefficients //take the array as pass by reference)
    ans = a 
    BEGIN for (i = 0, i < N+1, i++) 
        ans += coefficients[i] * (x ^ i)
    END for
END polynomial

BEGIN gaussian_error (input: stdev //we take this as input to scale the random distribution generated by box muller)
    u1, u2 = uniform random numbers between (0,1)
    z = sqrt(-2 * ln(u1)) * cos(2 * PI * u2) //applying box muller

    return z * stdev
END gaussian_error

BEGIN calc_root_mean_square_error (points, lagrange_points, newton_points)
    lagrange = 0
    newton = 0
    BEGIN for (i = 0, i < M, i++)
        lagrange += (points[i*2].y - lagrange_points[i]) ^ 2 
        newton += (points[i*2].y - newton_points[i]) ^ 2
    END for 

    lagrange = sqrt(lagrange / N)
    newton = sqrt(newton / N) 

    print("%N        %lagrange       %newton") 
END calc_root_mean_square_error

BEGIN Lagrange(input: points, lagrange_points //take these arrays as pass by reference)
    
    BEGIN for (i = 0, i < M, i++)

        //applying Lagrange fromula
        BEGIN for (j = 0, j < M, j++)
            x_j = points[2*j].x 
            term = points[2*j].y 
            
            BEGIN for (m = 0, m < M, m++)
                if (m != j) 
                    x_m = points[2*m].x 
                    term *= (points[2*i + 1].x - x_m) / (x_j - x_m)
                
            END for 
            
            lagrange_points[i] += term;
        END for 
    END for 
END Lagrange

BEGIN Newton(input: points, lagrange_points //take these arrays as pass by reference)
    
    BEGIN for (i = 0, i < M, i++)

        //applying Newton formula
        BEGIN for (j = 0, j < M, j++)
            term = 1

            BEGIN for (k = 0, k < j, k++)
                term *= (points[2*i + 1].x - points[2*k].x)
            END for 

            term *= divided_difference(points[2*j].x, points[0].x, points)
            lagrange_points[i] += term 
        END for
    END for 
END Newton

BEGIN divided_difference(input: start, stop, points //take the array as pass by reference)
    
    if ((start - stop) <= 1)
        return (points[start] - points[stop]) / (start - stop)

    return (divided_difference(start, stop + 1, points) - divided_difference(start - 1, stop, points)) / (start - stop)

END divided_difference
